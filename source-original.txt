
Тридцать первый · Серия 1 — код, часть 1: движок

Отдельный проект tridtsat-pervyi, не зависит от «Серии выживания». Части 1–3 вместе дают полный ролик: 90 с, 16 сцен (15 с голосом + титр), 200 слов, 1080×1920, 30 fps. Одно ядро на чистом JavaScript рисует кадр по его номеру: и HTML-превью в браузере, и Remotion для MP4.

Во всём проекте нет CSS-анимаций, transition, setTimeout и Math.random: всё движение считается от номера кадра.

Запуск

Создайте папку tridtsat-pervyi и разложите файлы из частей 1–3 по путям.

npm i remotion @remotion/cli react react-dom

Превью: npm run preview, затем откройте http://localhost:5050/preview/. Пробел — play/pause, стрелки — кадр, Shift+стрелки — секунда, ?frame=N — нужный кадр.

Remotion Studio: npm run studio.

Видео: npm run render → out/lift-ep1.mp4.

Стиль-лист: npm run sheet:a (Миша) и npm run sheet:b (элементы).

import {renderFrame, renderStyleSheet} from '/src/core/render.js';
import {SLOTS, STYLE_FRAMES, TOTAL_FRAMES, slotIndexAt, validateScript} from '/src/core/timeline.js';
import {FPS, H, W} from '/src/core/theme.js';
import {loadFonts} from '/src/core/fonts.js';

const $ = (id) => document.getElementById(id);
const stage = $('stage');
const box = $('frameBox');
const viewport = $('viewport');
const scrub = $('scrub');
const tc = $('tc');
const playBtn = $('play');
const modeSel = $('mode');
const issuesEl = $('issues');

const params = new URLSearchParams(location.search);
let mode = params.get('mode') === 'stylesheet' ? 'stylesheet' : 'episode';
let frame = Number(params.get('frame')) || 0;
let playing = false;
let t0 = 0;
let f0 = 0;
let drawn = -1;

const total = () => (mode === 'episode' ? TOTAL_FRAMES : STYLE_FRAMES);

function fit() {
  const s = Math.min(viewport.clientWidth / W, viewport.clientHeight / H);
  box.style.width = `${Math.floor(W * s)}px`;
  box.style.height = `${Math.floor(H * s)}px`;
  stage.style.transform = `scale(${s})`;
}

function label() {
  if (mode !== 'episode') return 'стиль-лист';
  const {def} = SLOTS[slotIndexAt(frame)];
  return def.kind === 'title' ? 'титр' : `сцена ${def.n}`;
}

function draw(force = false) {
  frame = Math.max(0, Math.min(total() - 1, Math.round(frame)));
  if (!force && frame === drawn) return;
  drawn = frame;
  stage.innerHTML = mode === 'episode' ? renderFrame(frame) : renderStyleSheet(frame);
  scrub.max = String(total() - 1);
  scrub.value = String(frame);
  tc.textContent = `кадр ${frame} / ${total() - 1} · ${(frame / FPS).toFixed(2)} с · ${label()}`;
}

function pause() {
  playing = false;
  playBtn.textContent = '▶';
}

function tick(now) {
  if (!playing) return;
  const f = f0 + Math.floor(((now - t0) / 1000) * FPS);
  if (f >= total() - 1) {
    frame = total() - 1;
    pause();
    draw();
    return;
  }
  frame = f;
  draw();
  requestAnimationFrame(tick);
}

function play() {
  if (frame >= total() - 1) frame = 0;
  playing = true;
  t0 = performance.now();
  f0 = frame;
  playBtn.textContent = '❚❚';
  requestAnimationFrame(tick);
}

function step(d) {
  pause();
  frame += d;
  draw();
}

playBtn.addEventListener('click', () => (playing ? pause() : play()));
$('prev').addEventListener('click', () => step(-1));
$('next').addEventListener('click', () => step(1));
scrub.addEventListener('input', () => {
  pause();
  frame = Number(scrub.value);
  draw();
});
modeSel.addEventListener('change', () => {
  pause();
  mode = modeSel.value;
  frame = 0;
  draw(true);
});
window.addEventListener('resize', fit);
window.addEventListener('keydown', (e) => {
  if (e.target === modeSel || e.target === scrub) return;
  if (e.code === 'Space') {
    e.preventDefault();
    if (playing) pause();
    else play();
  }
  if (e.code === 'ArrowRight') step(e.shiftKey ? FPS : 1);
  if (e.code === 'ArrowLeft') step(e.shiftKey ? -FPS : -1);
});

// Детерминированная математика анимации: одинаковый кадр = одинаковая картинка.
export const clamp = (v, a = 0, b = 1) => Math.min(b, Math.max(a, v));
export const mix = (a, b, t) => a + (b - a) * t;

export const Easing = {
  linear: (t) => t,
  inQuad: (t) => t * t,
  outQuad: (t) => 1 - (1 - t) * (1 - t),
  inOutQuad: (t) => (t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2),
  outCubic: (t) => 1 - Math.pow(1 - t, 3),
  inOutCubic: (t) => (t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2),
  outExpo: (t) => (t >= 1 ? 1 : 1 - Math.pow(2, -10 * t)),
};

// Аналог interpolate из Remotion.
export function interpolate(x, input, output, opts = {}) {
  if (input.length !== output.length || input.length < 2) {
    throw new Error('interpolate: диапазоны должны быть одной длины, минимум 2 точки');
  }
  const easing = opts.easing || Easing.linear;
  let i = 1;
  while (i < input.length - 1 && x > input[i]) i += 1;
  const x0 = input[i - 1];
  const x1 = input[i];
  let t = x1 === x0 ? 1 : (x - x0) / (x1 - x0);
  if (opts.extend !== true) t = clamp(t, 0, 1);
  return mix(output[i - 1], output[i], easing(t));
}

// Пружина как в Remotion: считается от номера кадра, без состояния.
export function spring(frame, {damping = 10, stiffness = 100, mass = 1, fps = 30, from = 0, to = 1} = {}) {
  if (frame <= 0) return from;
  const steps = Math.ceil(frame) * 4;
  const dt = frame / fps / steps;
  let x = from;
  let v = 0;
  for (let i = 0; i < steps; i += 1) {
    const a = (-stiffness * (x - to) - damping * v) / mass;
    v += a * dt;
    x += v * dt;
  }
  return x;
}

function hash(str) {
  let h = 2166136261;
  for (let i = 0; i < str.length; i += 1) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

// «Случайное» число 0…1 по сиду, вместо запрещённого Math.random.
export function random(seed) {
  let t = (hash(String(seed)) + 0x6d2b79f5) | 0;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
}

import {FONT_CSS_URL, FONT_PROBES, FONT_PROBE_TEXT} from './theme.js';

let pending = null;

// Подключает Google Fonts и ждёт загрузки кириллицы всех нужных начертаний.
export function loadFonts() {
  if (pending) return pending;
  pending = new Promise((resolve) => {
    if (typeof document === 'undefined') {
      resolve(false);
      return;
    }
    const waitFaces = () =>
      Promise.all(FONT_PROBES.map((f) => document.fonts.load(f, FONT_PROBE_TEXT)))
        .then(() => document.fonts.ready)
        .then(() => resolve(true))
        .catch(() => resolve(false));
    if (document.querySelector('link[data-episode-fonts]')) {
      waitFaces();
      return;
    }
    const link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = FONT_CSS_URL;
    link.setAttribute('data-episode-fonts', '1');
    link.onload = waitFaces;
    link.onerror = () => {
      console.warn('Шрифты Google Fonts не загрузились: проверьте интернет');
      resolve(false);
    };
    document.head.appendChild(link);
  });
  return pending;
}

src/core/timeline.js

// Таймлайн серии: слоты сцен, мягкие склейки, расчётный тайминг слов, проверка сценария.
import {EPISODE, SCENES} from './script.js';
import {FPS} from './theme.js';
import {clamp, Easing} from './math.js';

export const XFADE = 10; // мягкая склейка сцен: 0,33 с
export const TITLE_FADE = 12; // вход и выход чёрного титра: 0,4 с
export const VOICE_PAD = 9; // реплика: через 0,3 с после начала сцены, до 0,3 с до конца
export const STYLE_FRAMES = 90;

export const SLOTS = [];
let cursor = 0;
SCENES.forEach((def, index) => {
  const frames = Math.round(def.dur * FPS);
  SLOTS.push({def, index, from: cursor, frames});
  cursor += frames;
});
export const TOTAL_FRAMES = cursor; // 2700 кадров = 90 с

export function slotIndexAt(frame) {
  const f = clamp(Math.floor(frame), 0, TOTAL_FRAMES - 1);
  let found = 0;
  for (let i = 0; i < SLOTS.length; i += 1) {
    if (f >= SLOTS[i].from) found = i;
  }
  return found;
}

const fadeBetween = (a, b) => (a.def.kind === 'title' || b.def.kind === 'title' ? TITLE_FADE : XFADE);

// Слои кадра снизу вверх. На склейке новая сцена проявляется поверх предыдущей.
export function layersAt(frame) {
  const f = clamp(Math.floor(frame), 0, TOTAL_FRAMES - 1);
  const i = slotIndexAt(f);
  const cur = SLOTS[i];
  const local = f - cur.from;
  if (i > 0) {
    const prev = SLOTS[i - 1];
    const n = fadeBetween(prev, cur);
    if (local < n) {
      return [
        {slot: prev, local: f - prev.from, opacity: 1},
        {slot: cur, local, opacity: Easing.inOutQuad((local + 1) / (n + 1))},
      ];
    }
  }
  return [{slot: cur, local, opacity: 1}];
}

const HAS_LETTER = /[0-9A-Za-zА-Яа-яЁё]/;
const VOWELS = /[аеёиоуыэюяaeiouy]/g;

// Слова реплики. Тире и одиночные знаки приклеиваются к предыдущему слову и дают паузу.
export function tokenize(voice = '') {
  const words = [];
  voice
    .split(' ')
    .filter(Boolean)
    .forEach((raw) => {
      if (!HAS_LETTER.test(raw)) {
        const last = words[words.length - 1];
        if (last) {
          last.text += ` ${raw}`;
          last.pause += 1;
        }
        return;
      }
      const clean = raw.toLowerCase().replace(/[^0-9a-zа-яё]/g, '');
      const syl = Math.max(1, (clean.match(VOWELS) || []).length);
      const pause = /[.!?…]$/.test(raw) ? 1.2 : /[,:;]$/.test(raw) ? 0.6 : 0;
      words.push({text: raw, clean, syl, pause});
    });
  return words;
}

export const wordCount = (voice = '') => tokenize(voice).length;

// Расчётный тайминг слов, пока нет озвучки: длина слова по слогам плюс паузы на знаках.
export function wordTimings(def, frames) {
  const words = tokenize(def.voice);
  if (!words.length) return [];
  const a = VOICE_PAD;
  const b = frames - VOICE_PAD;
  const units = words.reduce((s, w, i) => s + w.syl + 0.5 + (i < words.length - 1 ? w.pause : 0), 0);
  const k = (b - a) / units;
  let t = a;
  return words.map((w, i) => {
    const start = t;
    t += (w.syl + 0.5) * k;
    const end = t;
    if (i < words.length - 1) t += w.pause * k;
    return {...w, start, end};
  });
}

// Проверка: длина 90 с, сцены 4–7 с, темп 2,4–2,7 слова/с, всего 200 слов.
export function validateScript() {
  const issues = [];
  const totalSec = TOTAL_FRAMES / FPS;
  if (Math.abs(totalSec - EPISODE.targetSeconds) > 0.01) issues.push(`длина ${totalSec} с вместо ${EPISODE.targetSeconds} с`);
  let words = 0;
  SLOTS.forEach(({def}) => {
    if (def.kind !== 'scene') return;
    const n = wordCount(def.voice);
    words += n;
    const rate = n / (def.dur - (2 * VOICE_PAD) / FPS);
    if (def.dur < EPISODE.minScene || def.dur > EPISODE.maxScene) issues.push(`${def.id}: ${def.dur} с вне ${EPISODE.minScene}–${EPISODE.maxScene} с`);
    if (rate < EPISODE.minRate - 1e-6 || rate > EPISODE.maxRate + 1e-6) issues.push(`${def.id}: темп ${rate.toFixed(2)} сл/с`);
  });
  if (words !== EPISODE.targetWords) issues.push(`слов ${words} вместо ${EPISODE.targetWords}`);
  return issues;
}

// Караоке-субтитры: показывается текущая фраза (1–2 строки). Несказанные слова серые,
// сказанные белые, ключевые — цветом смысла со свечением.
import {C, FONT_HEAVY, colorOf, glow} from './theme.js';
import {Easing, clamp, interpolate, spring} from './math.js';
import {wordTimings} from './timeline.js';
import {esc, f2, f3} from './html.js';

const SHADOW = '0 6px 0 rgba(0,0,0,0.55), 0 0 18px rgba(0,0,0,0.85)';
const MAX = 38;
const END = /[.!?…:][»"]?$/;
const SOFT = /[,—]$/;

// Фразы: конец предложения, длинная часть до запятой или тире, не больше 38 знаков
// (последнее слово предложения может немного превысить, чтобы не оставлять его одно).
function chunksOf(words) {
  const out = [];
  let cur = [];
  let len = 0;
  const flush = () => {
    if (cur.length) out.push(cur);
    cur = [];
    len = 0;
  };
  words.forEach((w, i) => {
    const add = w.text.length + (cur.length ? 1 : 0);
    const ends = END.test(w.text);
    if (cur.length && len + add > MAX && !(ends && len + add <= MAX + 10)) flush();
    len += w.text.length + (cur.length ? 1 : 0);
    cur.push(i);
    if (ends || (SOFT.test(w.text) && len >= 18)) flush();
  });
  flush();
  return out;
}

export function subtitles(def, local, frames) {
  if (!def.voice) return '';
  const words = wordTimings(def, frames);
  if (!words.length) return '';
  const first = words[0].start;
  if (local < first - 4 || local > frames - 2) return '';
  let cur = 0;
  words.forEach((w, i) => {
    if (local >= w.start) cur = i;
  });
  const chunks = chunksOf(words);
  const ch = chunks.find((c) => c.includes(cur)) || chunks[0];
  const isLast = ch === chunks[chunks.length - 1];
  const fade = isLast ? interpolate(local, [frames - 7, frames - 2], [1, 0]) : 1;
  const a = clamp(spring(local - words[ch[0]].start + 3, {damping: 14, stiffness: 200}), 0, 1.1);
  const spans = ch.map((i) => {
    const w = words[i];
    const said = local >= w.start;
    const key = def.keys ? def.keys[w.clean] : null;
    const kc = key ? colorOf(key) : null;
    const color = !said ? '#6E7486' : kc || C.white;
    const k = said ? clamp((local - w.start) / 6) : 1;
    const sc = said ? 1 + 0.12 * (1 - Easing.outCubic(k)) : 1;
    const sh = said && kc ? `${glow(kc, 0.7)}, 0 6px 0 rgba(0,0,0,0.55)` : SHADOW;
    return `<span style='display:inline-block;color:${color};text-shadow:${sh};transform:scale(${f3(sc)})'>${esc(w.text)}</span>`;
  });
  return `<div style='position:absolute;left:44px;right:44px;top:1452px;text-align:center;font-family:${FONT_HEAVY};font-weight:900;font-size:76px;line-height:1.1;text-transform:uppercase;opacity:${f3(fade * clamp(a * 1.3))};transform:translateY(${f2((1 - Math.min(1, a)) * 16)}px)'>${spans.join(' ')}</div>`;
}

// Наезд 1 → 1.08 за сцену, тряска только на ударах.
export function cameraTransform(impacts, local, frames) {
  const s = interpolate(local, [0, frames], [1, 1.08], {easing: Easing.inOutQuad});
  let dx = 0;
  let dy = 0;
  impacts.forEach((t) => {
    const k = local - t;
    if (k >= 0 && k < 16) {
      const a = 12 * (1 - k / 16);
      dx += Math.sin(k * 2.7) * a;
      dy += Math.cos(k * 3.3) * a * 0.7;
    }
  });
  return `translate(${f2(dx)}px, ${f2(dy)}px) scale(${s.toFixed(5)})`;
}

// Расходящиеся пунктирные круги звука. Круг i стартует со сдвигом, чтобы не появляться сразу большим.
export function soundRings(cx, cy, local, o = {}) {
  const {color = C.red, n = 3, period = 30, maxR = 360, minR = 20, width = 6, dash = '22 14', from = 0, opacity = 0.8} = o;
  const t0 = local - from;
  if (t0 < 0) return '';
  let out = '';
  for (let i = 0; i < n; i += 1) {
    const off = (i * period) / n;
    if (t0 < off) continue;
    const ph = ((t0 - off) % period) / period;
    out += `<circle cx='${f2(cx)}' cy='${f2(cy)}' r='${f2(minR + ph * (maxR - minR))}' fill='none' stroke='${color}' stroke-width='${width}' stroke-dasharray='${dash}' opacity='${f3((1 - ph) * opacity)}'/>`;
  }
  return out;
}

export function dust(global, n = 30) {
  let c = '';
  for (let i = 0; i < n; i += 1) {
    const sp = 0.2 + random('ds' + i) * 0.6;
    const y = (((random('dy' + i) * H - global * sp) % H) + H) % H;
    const x = random('dx' + i) * W + Math.sin(global / 40 + i) * 8;
    const o = 0.08 + 0.25 * (0.5 + 0.5 * Math.sin(global / 25 + i * 1.7));
    c += `<circle cx='${f2(x)}' cy='${f2(y)}' r='${f2(1.2 + random('dr' + i) * 2.6)}' fill='#C9CEDD' opacity='${f3(o)}'/>`;
  }
  return `<svg width='${W}' height='${H}' style='position:absolute;left:0;top:0'>${c}</svg>`;
}

export function mixPose(a, b, t) {
  const m = (p, q) => [mix(p[0], q[0], t), mix(p[1], q[1], t)];
  return {
    la: m(a.la, b.la), ra: m(a.ra, b.ra), ll: m(a.ll, b.ll), rl: m(a.rl, b.rl),
    thigh: mix(a.thigh, b.thigh, t), shin: mix(a.shin ?? SHIN, b.shin ?? SHIN, t),
  };
}
function resolvePose(pose, pose2, mixT) {
  const a = POSES[pose] || POSES.stand;
  return pose2 ? mixPose(a, POSES[pose2] || POSES.stand, mixT) : a;
}
function groundOffset(P) {
  return -10 + P.thigh * Math.cos(rad(P.ll[0])) + (P.shin ?? SHIN) * Math.cos(rad(P.ll[0] + P.ll[1])) + 28;
}
function blinkLid(base, local, seed) {
  const period = 96 + ((seed * 17) % 40);
  const bt = (((local + seed * 29) % period) + period) % period;
  const blink = bt < 3 ? bt / 3 : bt < 6 ? (6 - bt) / 3 : 0;
  return Math.min(1, base + blink * (1 - base));
}

function eye(cx, cy, r, lid, px, py, pupil) {
  if (lid >= 0.97) {
    return `<circle cx='${cx}' cy='${cy}' r='${r + 1}' fill='${M.skin}'/><path d='M ${cx - r + 4} ${cy + 2} Q ${cx} ${cy + 12} ${cx + r - 4} ${cy + 2}' stroke='${M.line}' stroke-width='5' fill='none' stroke-linecap='round'/>`;
  }
  let out = `<circle cx='${cx}' cy='${cy}' r='${r}' fill='${M.white}'/><circle cx='${f2(cx + px)}' cy='${f2(cy + 3 + py)}' r='${pupil}' fill='${M.pupil}'/><circle cx='${f2(cx + px + 4)}' cy='${f2(cy - 2 + py)}' r='${f2(pupil * 0.3)}' fill='${M.white}'/>`;
  if (lid > 0.02) {
    const R = r + 2; // веко чуть больше глаза: кольцо белка не просвечивает
    const dy = -r + 2 * r * lid;
    const w = Math.sqrt(Math.max(0, R * R - dy * dy));
    const lw = Math.sqrt(Math.max(0, r * r - dy * dy));
    out += `<path d='M ${f2(cx - w)} ${f2(cy + dy)} A ${R} ${R} 0 ${dy > 0 ? 1 : 0} 1 ${f2(cx + w)} ${f2(cy + dy)} Z' fill='${M.skin}'/>`;
    out += `<line x1='${f2(cx - lw)}' y1='${f2(cy + dy)}' x2='${f2(cx + lw)}' y2='${f2(cy + dy)}' stroke='${M.skinSh}' stroke-width='4' stroke-linecap='round'/>`;
  }
  return out;
}

// Точка на фигуре в координатах кадра: top (макушка), head (центр головы), mouth (рот).

}

// Разрез башни: этажи from…to снизу вверх, шахта, кабина (зелёная), красные точки — заражённые.
export const floorTop = (o, n) => o.ground - (n - (o.from ?? 1) + 1) * o.fh;

export function towerSection(uid, o) {
  const {x, w, ground, fh, from = 1, to = 31, shaftX, sw = 90, cabinFloor = 31, dots = {}, labels = [], roof = true} = o;
  const fy = (n) => ground - (n - from + 1) * fh;
  const topY = fy(to);
  let s = `<rect x='${x}' y='${f2(topY)}' width='${w}' height='${f2(ground - topY)}' fill='#1A1F36'/>`;
  for (let n = from; n <= to; n += 1) {
    const yy = fy(n);
    s += `<line x1='${x}' y1='${f2(yy)}' x2='${x + w}' y2='${f2(yy)}' stroke='#2A3050' stroke-width='3'/>`;
    if (labels.includes(n)) s += svgText(x - 14, yy + fh * 0.5 + 8, String(n), {size: 22, anchor: 'end', color: n === cabinFloor ? C.green : C.grey, weight: 700});
    const d = dots[n] || 0;
    if (d > 0.01) {
      [0.18, 0.82].forEach((u) => {
        const dx = x + u * w;
        if (dx > shaftX - 14 && dx < shaftX + sw + 14) return;
        s += `<circle cx='${f2(dx)}' cy='${f2(yy + fh / 2)}' r='${f2(Math.min(9, fh * 0.32))}' fill='${C.red}' opacity='${f3(d)}'/>`;
      });
    }
  }
  s += `<rect x='${shaftX}' y='${f2(topY)}' width='${sw}' height='${f2(ground - topY)}' fill='#0A0C15'/>`;
  const cy = fy(cabinFloor);
  s += `<rect x='${shaftX + 5}' y='${f2(cy + 3)}' width='${sw - 10}' height='${f2(fh - 6)}' rx='4' fill='${C.green}' fill-opacity='0.3' stroke='${C.green}' stroke-width='4'/>`;
  if (roof) s += `<rect x='${x - 16}' y='${f2(topY - 18)}' width='${w + 32}' height='18' fill='#2E3658'/>`;
  return s;
}

// 13 · Поворот: из шахты стучат ··· ——— ··· (SOS). Это человек.
export function s13({local, uid}) {
  const seq = [['d', 12], ['d', 20], ['d', 28], ['l', 46], ['l', 62], ['l', 78], ['d', 96], ['d', 104], ['d', 112]];
  const y = 930;
  let x = 170;
  const items = seq.map(([t, at], i) => {
    const w = t === 'd' ? 36 : 84;
    const cx = x + w / 2;
    x += w + (i === 2 || i === 5 ? 64 : 24);
    return {t, at, cx};
  });
  let svg = cabin(uid, {floor: '31'}) + cabinLight(0.3);
  svg += `<rect x='120' y='800' width='840' height='280' rx='30' fill='rgba(7,8,15,0.9)' stroke='${C.gold}' stroke-opacity='0.4' stroke-width='3'/>`;
  items.forEach((it) => {
    if (local < it.at) {
      svg += it.t === 'd'
        ? `<circle cx='${f2(it.cx)}' cy='${y}' r='18' fill='#2A2F45'/>`
        : `<rect x='${f2(it.cx - 42)}' y='${y - 12}' width='84' height='24' rx='12' fill='#2A2F45'/>`;
      return;
    }
    const a = Math.min(1.2, spring(local - it.at, {damping: 12, stiffness: 220}));
    const sc = 0.6 + 0.4 * a;
    const shape = it.t === 'd' ? `<circle r='18' fill='${C.gold}'/>` : `<rect x='-42' y='-12' width='84' height='24' rx='12' fill='${C.gold}'/>`;
    svg += `<g transform='translate(${f2(it.cx)} ${y}) scale(${f3(sc)})'>${shape}</g>`;
    const k = local - it.at;
    if (k < 14) svg += svgText(it.cx, y - 70, 'ТУК', {font: 'heavy', size: 30, color: '#FFFFFF', weight: 900, anchor: 'middle', spacing: 0, opacity: 1 - k / 14});
  });
  [[0, 2, 30], [3, 5, 80], [6, 8, 114]].forEach(([a, b, t], gi) => {
    const cx = (items[a].cx + items[b].cx) / 2;
    svg += svgText(cx, y + 100, gi === 1 ? 'O' : 'S', {font: 'heavy', size: 64, color: C.gold, weight: 900, anchor: 'middle', spacing: 0, opacity: L(local, t, t + 8)});
  });
  const cam = hud({x: 300, y: 1140, color: 'gold', icon: 'target', title: 'SOS', sub: 'сигнал бедствия', local, delay: 118});
  return {svg, cam};
}
